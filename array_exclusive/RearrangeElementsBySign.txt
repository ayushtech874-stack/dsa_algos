Solution TYPE 1 : 
Bruteforce : We will iterate through array and store positive and negative in seperate arrays

we will put -ve elements TC = 3N/2   SC = N

Optimal : We cannot optimise the SC but we can do this for time complexity 

Pseudocode : 

int n = nums.size();
vector<int> ans(n,0); // answer vector list
int pos = 0 ; // we will store pos at first at 0th index
int neg = 1 ; // we will store neg at first at 1st index
for(int i = 0 ; i<n ; i++)
{
    if(nums[i] > 0)
    {
        nums[pos] = nums[i];
        pos + = 2;
    }
    else 
    {
        nums[neg] = nums[i];

}

/*

-ve and that last whatever left will store it directly in same order 

PseudoCode : 

vector<int> pos;
vector<int> neg;
int n = nums.size();
for(int i = 0 ; i<n ; i++)
{
    if(nums[i]>0)
    {
        pos.push_back(nums[i]);
    }
    else 
    {
        neg.push_back(nums[i]);
    }
}
if(pos.size()  > neg.size())
{
    // Positive is more negative is less 
    for(int i = 0 ; i<neg.size(); i++)
    {
        nums[2*i] = pos[i];
        nums[2*i+1] = neg[i]; 
    }

    int index = neg.size() * 2;

    // eg neg.size()= 2   , so we are going to store 2 +ve and 2 -ve so next time we are 
    // going to store directly from index 4 all the positive numbers

    for(int i = neg.size() ; i<pos.size(); i++)  // iterating positive list
    {
        nums[index] = pos[i];
        index++;
    }
}
else
{
    // Negative is more Positive is less 
    for(int i = 0 ; i<pos.size(); i++)
    {
        nums[2*i] = pos[i];
        nums[2*i+1] = neg[i]; 
    }

    int index = pos.size() * 2;

    // eg pos.size()= 2   , so we are going to store 2 +ve and 2 -ve so next time we are 
    // going to store directly from index 4 all the negative numbers

    for(int i = pos.size() ; i < neg.size(); i++)  // iterating negative list
    {
        nums[index] = neg[i];
        index++;
    }
}
return nums;
